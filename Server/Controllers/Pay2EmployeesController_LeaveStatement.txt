        [HttpGet("{empId}/leave-statement/{year}/pdf")]
        public async Task<IActionResult> DownloadLeaveStatementPdf(int empId, int year, [FromServices] IPay2AccessService acl)
        {
            var userco = int.Parse(User.FindFirstValue(ClaimTypes.NameIdentifier) ?? "0");

            // 1. Fetch Employee Info to check WS Access
            var empQuery = "SELECT EMP_ID, EMP_CODE, WS_ID, FIRST_NAME, LAST_NAME, USERCO FROM PAY2_EMPLOYEE WHERE EMP_ID = @empId";
            var emp = await _db.QueryFirstOrDefaultAsync<dynamic>(empQuery, new { empId });
            if (emp == null) return NotFound("Employee not found");

            // 2. ACL Check: Self-Service OR Manager with Read Access
            if (emp.USERCO != userco)
            {
                await acl.EnsureWorkshopAsync(userco, Pay2Perm.See, Pay2Form.Pay2Employee, (int)emp.WS_ID);
            }

            // 3. Gather Config Data
            var configQuery = @"
                SELECT
                    ISNULL(MAX(CASE WHEN CFG_KEY = 'LEAVE_MINS_PER_DAY' THEN TRY_CAST(CFG_VALUE AS INT) END), 440) AS LeaveMinsPerDay,
                    ISNULL(MAX(CASE WHEN CFG_KEY = 'LEAVE_CARRYOVER_MAX' THEN TRY_CAST(CFG_VALUE AS INT) END), 9) AS LeaveCarryoverMax
                FROM PAY2_CONFIG
                WHERE CFG_KEY IN ('LEAVE_MINS_PER_DAY', 'LEAVE_CARRYOVER_MAX')";
            var config = await _db.QueryFirstOrDefaultAsync<dynamic>(configQuery);

            // 4. Gather Balance Data
            var balQuery = "SELECT ENTITLEMENT_MIN, USED_MIN, CARRIED_IN_MIN FROM PAY2_LEAVE_BAL WHERE EMP_ID = @empId AND YEAR = @year";
            var bal = await _db.QueryFirstOrDefaultAsync<dynamic>(balQuery, new { empId, year });

            int entitlement = bal?.ENTITLEMENT_MIN ?? (config.LeaveMinsPerDay * 26);
            int carriedIn = bal?.CARRIED_IN_MIN ?? 0;
            int used = bal?.USED_MIN ?? 0;
            int balance = entitlement + carriedIn - used;

            // 5. Gather History
            var historyQuery = @"
                SELECT
                    START_DATE,
                    END_DATE,
                    REQ_DAYS,
                    REQ_HOURS,
                    REQ_MINUTES,
                    DESCRIPTION,
                    (REQ_DAYS * @LeaveMinsPerDay + REQ_HOURS * 60 + REQ_MINUTES) AS TotalDeductedMinutes
                FROM PAY2_LEAVE
                WHERE EMP_ID = @empId
                  AND START_DATE / 10000 = @year
                  AND STATUS >= 3
                ORDER BY START_DATE DESC";

            var history = (await _db.QueryAsync<Pay2LeaveStatementLineDto>(historyQuery, new { empId, year, LeaveMinsPerDay = config.LeaveMinsPerDay })).ToList();

            var dto = new Pay2LeaveStatementDto
            {
                EmployeeCode = emp.EMP_CODE,
                EmployeeName = $"{emp.LAST_NAME} {emp.FIRST_NAME}",
                Year = year,
                PrintDate = Safir.Shared.Helpers.PersianDateHelper.GetPersianDate(DateTime.Now),
                LeaveMinsPerDay = config.LeaveMinsPerDay,
                LeaveCarryoverMax = config.LeaveCarryoverMax,
                EntitlementMin = entitlement,
                CarriedInMin = carriedIn,
                UsedMin = used,
                BalanceMin = balance,
                History = history
            };

            // 6. Generate PDF using QuestPDF
            var doc = new Safir.Server.Reports.Pay2LeaveStatementDocument(dto);
            byte[] pdfBytes = doc.GeneratePdf();

            return File(pdfBytes, "application/pdf", $"LeaveStatement_{emp.EMP_CODE}_{year}.pdf");
        }
